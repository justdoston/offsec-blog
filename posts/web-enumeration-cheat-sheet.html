Directory brute forcing

Directory brute forcing with gobuster

# Basic directory scan:
gobuster dir -u http://target.com -w /usr/share/wordlists/dirbuster/directory-list-2.3-medium.txt

# With extensions:
gobuster dir -u http://target.com -w wordlist.txt -x php,html,txt

# With status codes:
gobuster dir -u http://target.com -w wordlist.txt -s 200,204,301,302,307,401,403

# Recursive scan:
gobuster dir -u http://target.com -w wordlist.txt -r

# With threads:
gobuster dir -u http://target.com -w wordlist.txt -t 50

# Exclude status codes:
gobuster dir -u http://target.com -w wordlist.txt -b 404,403

Directory discovery autmation

# Bash script:
#!/bin/bash
wordlist="/usr/share/wordlists/dirbuster/directory-list-2.3-medium.txt"
target="http://target.com"

while IFS= read -r dir; do
    response=$(curl -s -o /dev/null -w "%{http_code}" "$target/$dir")
    if [ "$response" != "404" ]; then
        echo "[$response] $target/$dir"
    fi
done < "$wordlist"

# Python script:
import requests
from concurrent.futures import ThreadPoolExecutor

def check_dir(url, directory):
    try:
        r = requests.get(f"{url}/{directory}", timeout=5)
        if r.status_code != 404:
            print(f"[{r.status_code}] {url}/{directory}")
    except:
        pass

with open("wordlist.txt") as f:
    dirs = [line.strip() for line in f]

with ThreadPoolExecutor(max_workers=50) as executor:
    executor.map(lambda d: check_dir("http://target.com", d), dirs)
